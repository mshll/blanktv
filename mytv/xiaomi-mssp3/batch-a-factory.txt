# mitv.service, hotkey dispatcher, remote service, Netflix (and netflixobserver, which the remote's Netflix button needs).

# PatchWall, content rows, sponsored rows

com.xiaomi.floatingframe

com.xiaomi.mitv.tvmanager                 # "TV Manager" cleaner, ~48 MB; kills background apps
com.xiaomi.mitv.updateservice             # firmware updates: re-enable to check for one

com.google.android.katniss                # Google Assistant; comes back at boot (see the example's header)

com.google.android.videos                 # Play Movies & TV (already disabled on this firmware)
